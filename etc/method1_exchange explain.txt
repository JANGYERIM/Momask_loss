■ 새로 추가된 메서드: _load_teacher_caption

    def _load_teacher_caption(self, name):
        if self.teacher_text_dir is None:
            return None

        # 서브세그먼트 이름 처리
        # 원본 데이터에서 시간 범위가 있는 모션은 "A_000001" 형태로 저장됨
        # (A~W 랜덤 prefix 1글자 + '_' + 원본 이름)
        # teacher caption 파일은 원본 이름 기준이므로 prefix 제거 필요
        base_name = name.split('_', 1)[1] if (len(name) > 2 and name[1] == '_') else name

        teacher_path = pjoin(self.teacher_text_dir, base_name + '.txt')
        try:
            with open(teacher_path, 'r') as f:
                return f.read().strip()
        except:
            return None    # 파일 없으면 None → __getitem__에서 caption으로 대체

  → 서브세그먼트 이름 예시:
      data_dict 키: "C_000001"  (C가 랜덤 prefix)
      실제 모션 파일: "000001.npy"
      teacher caption 파일: teacher_text_dir/000001.txt

  → try/except 로 파일이 없어도 조용히 None 반환.
    teacher caption 생성이 일부 완료된 상태에서도 학습 가능.

    def forward(self, ids, y, m_lens):

        bs, ntokens = ids.shape
        device = ids.device

        # 패딩 처리
        non_pad_mask = lengths_to_mask(m_lens, ntokens)
        ids = torch.where(non_pad_mask, ids, self.pad_id)

        # 텍스트 인코딩
        force_mask = False
        if self.cond_mode == 'text':
            with torch.no_grad():
                cond_vector = self.encode_text(y)    # CLIP 인코딩
        ...

        # 마스크 샘플링
        rand_time = uniform((bs,), device=device)
        rand_mask_probs = self.noise_schedule(rand_time)
        num_token_masked = (ntokens * rand_mask_probs).round().clamp(min=1)
        batch_randperm = torch.rand((bs, ntokens), device=device).argsort(dim=-1)
        mask = batch_randperm < num_token_masked.unsqueeze(-1)
        mask &= non_pad_mask

        labels = torch.where(mask, ids, self.mask_id)

        x_ids = ids.clone()
        # BERT 마스킹 (10% 랜덤 토큰, 88% MASK 토큰)
        mask_rid = get_mask_subset_prob(mask, 0.1)
        rand_id = torch.randint_like(x_ids, high=self.opt.num_tokens)
        x_ids = torch.where(mask_rid, rand_id, x_ids)
        mask_mid = get_mask_subset_prob(mask & ~mask_rid, 0.88)
        x_ids = torch.where(mask_mid, self.mask_id, x_ids)

        # forward + loss
        logits = self.trans_forward(x_ids, cond_vector, ~non_pad_mask, force_mask)
        ce_loss, pred_id, acc = cal_performance(logits, labels, ignore_index=self.mask_id)

        return ce_loss, pred_id, acc

    def _sample_mask(self, ids, m_lens):

        device = ids.device

        non_pad_mask = lengths_to_mask(m_lens, ntokens)
        ids = torch.where(non_pad_mask, ids, self.pad_id)

        rand_time = uniform((bs,), device=device)
        rand_mask_probs = self.noise_schedule(rand_time)
        num_token_masked = (ntokens * rand_mask_probs).round().clamp(min=1)

        batch_randperm = torch.rand((bs, ntokens), device=device).argsort(dim=-1)
        mask = batch_randperm < num_token_masked.unsqueeze(-1)
        mask &= non_pad_mask

        labels = torch.where(mask, ids, self.mask_id)

        x_ids = ids.clone()

        rand_id = torch.randint_like(x_ids, high=self.opt.num_tokens)
        x_ids = torch.where(mask_rid, rand_id, x_ids)
        mask_mid = get_mask_subset_prob(mask & ~mask_rid, 0.88)
        x_ids = torch.where(mask_mid, self.mask_id, x_ids)

  [수정된 forward()]

    def forward(self, ids, y, m_lens, teacher_y=None):

        bs, ntokens = ids.shape
        device = ids.device

        # 마스크 샘플링 (분리된 메서드 호출)
        x_ids, labels, non_pad_mask = self._sample_mask(ids, m_lens)

        # 원본 캡션 인코딩
        force_mask = False
        if self.cond_mode == 'text':
            with torch.no_grad():
                cond_vector = self.encode_text(y)
        ...

        # 원본 캡션으로 forward
        logits = self.trans_forward(x_ids, cond_vector, ~non_pad_mask, force_mask)

        if teacher_y is not None:
            # teacher 캡션 인코딩 (같은 x_ids, 같은 labels 사용)
            with torch.no_grad():
                teacher_cond_vector = self.encode_text(teacher_y)
            # teacher 캡션으로 forward (x_ids는 동일 → 같은 마스크 위치 예측)
            teacher_logits = self.trans_forward(x_ids, teacher_cond_vector, ~non_pad_mask, force_mask)

            # 두 loss를 각각 계산하고 합산
            loss1, pred_id, acc = cal_performance(logits, labels, ignore_index=self.mask_id)
            loss2, _, _         = cal_performance(teacher_logits, labels, ignore_index=self.mask_id)
            ce_loss = loss1 + loss2
        else:
            # teacher_y 없을 때는 기존과 동일하게 동작
            ce_loss, pred_id, acc = cal_performance(logits, labels, ignore_index=self.mask_id)

        return ce_loss, pred_id, acc

■ 핵심 포인트: 왜 _sample_mask를 분리했는가?
-----------------------------------------------
  원본 forward()에서는 마스크 샘플링과 텍스트 인코딩이 뒤섞여 있었다.
  teacher caption을 쓰려면 "같은 마스크 위치"를 두 캡션이 공유해야 하는데,
  마스크 샘플링이 forward() 내부에 있으면 두 번 호출 시 마스크가 달라진다.

  따라서:
    - _sample_mask() : 마스크 위치 결정 (1번만 실행)
    - trans_forward() : 해당 마스크 + 텍스트 조건으로 예측 (캡션마다 실행)

  이렇게 분리하면 두 캡션이 "정확히 같은 토큰 위치"를 예측하게 된다.

■ BERT 마스킹 스킴이란?
--------------------------
  MoMask는 단순히 mask_id를 채우는 것이 아니라 BERT 방식의 마스킹을 적용:
    - 마스킹 대상 토큰의 88% → mask_id 로 교체 (모델이 복원해야 함)
    - 마스킹 대상 토큰의 10% → 랜덤 다른 토큰으로 교체 (노이즈 주입)
    - 나머지 2%           → 원래 토큰 유지 (Identity)
  labels는 항상 "원래 정답 토큰"을 담고 있어서 loss 계산은 labels 기준으로 함.

■ trans_forward 란?
----------------------
  trans_forward(x_ids, cond_vector, padding_mask, force_mask)
    - x_ids: 마스킹이 적용된 토큰 시퀀스 (일부는 mask_id, 일부는 랜덤, 나머지는 원본)
    - cond_vector: CLIP 인코딩된 텍스트 임베딩 (b, 512)
    - padding_mask: 패딩 위치 표시 (True = 패딩)
    - 출력 logits: (b, num_tokens, seqlen) — 각 위치의 토큰 확률 분포

  즉, "이 텍스트 조건으로 마스킹된 위치들을 예측하라"는 연산.

  transformer 내부에서 같은 마스크로 두 번 trans_forward를 실행한다.

■ 원본과의 계산량 차이
================================================================================

  원본:
    - CLIP 인코딩: 1회 (원본 캡션)
    - trans_forward: 1회
    - cal_performance: 1회

  _tc 버전:
    - CLIP 인코딩: 2회 (원본 + teacher)
    - trans_forward: 2회
    - cal_performance: 2회
    - 마스크 샘플링: 1회 (분리되어 한 번만 실행)

  → 스텝당 약 2배 연산. CLIP은 frozen + no_grad 이므로 메모리 증가는 크지 않음.
    trans_forward 2회가 주요 오버헤드.

■ 향후 확장 방향 (현재 미구현)
================================================================================

  현재 구현:
    ce_loss = loss1 + loss2  (단순 합산)

  향후 확장 (방법 C 심화):
    - pred1 = argmax(logits1), pred2 = argmax(logits2)
    - both_wrong = (pred1 != labels) & (pred2 != labels)
    - 두 캡션 모두 틀린 위치에 loss weight 상향 (예: ×2)
    - → 모델이 "어려운 토큰"에 집중하도록 유도
    - 구현 위치: tools.py 에 cal_performance_weighted() 추가
